+"""A settlement that does not add up to itself.
+
+The reported incident: one settlement's reported net was INR 250.00 below what
+its own gross, fees, tax and refunds add up to. Fees and tax were exactly on the
+fee schedule, and no record explained the 250.00. It was closed automatically as
+a fee mismatch.
+
+Two defects produced that. The fee-mismatch residual was rebuilt from the
+settlement's components instead of measured against its reported net, so the
+contradiction was invisible to it (residual 0). And a broken arithmetic row was
+labelled a fee mismatch even though the fee schedule check passed.
+
+These tests pin the behaviour that follows from the project's rule that money
+is only closed when a proven cause accounts for all of it.
+"""
+
+from __future__ import annotations
+
+from datetime import datetime, timedelta
+
+from ledgerguard.ai.provider import HeuristicProvider
+from ledgerguard.evidence.safety import AUTO_RESOLVED, HUMAN_REVIEW_REQUIRED
+from ledgerguard.evidence.verifier import VERIFIED
+from ledgerguard.ledger.models import (
+    BankEntry,
+    Batch,
+    CreditDebit,
+    ExceptionType,
+    Order,
+    Payment,
+    Refund,
+    Settlement,
+)
+from ledgerguard.ledger.money import gst_on_fee, platform_fee, to_paise
+from ledgerguard.pipeline import run
+from ledgerguard.reconciliation.matcher import reconcile_batch
+
+T0 = datetime(2026, 7, 15, 10, 0, 0)
+SHORT = to_paise("250.00")
+
+
+def _batch(
+    *,
+    amount: int = to_paise("5000.00"),
+    fee_overcharge: int = 0,
+    short: int = 0,
+    orphan_refund_amount: int | None = None,
+    customer: str = "CUST-T1",
+) -> Batch:
+    """One lifecycle. The settlement's reported net is ``short`` below its components."""
+    order = Order(
+        order_id="ord_T1",
+        merchant_id="mch_T1",
+        amount=amount,
+        created_at=T0,
+        customer_reference=customer,
+    )
+    fee = platform_fee(amount)
+    tax = gst_on_fee(fee)
+    captured = T0 + timedelta(minutes=5)
+    payment = Payment(
+        payment_id="pay_T1",
+        order_id=order.order_id,
+        amount=amount,
+        captured_at=captured,
+        fee=fee,
+        tax=tax,
+        reference="PAY/ord_T1",
+    )
+    charged_fee = fee + fee_overcharge
+    charged_tax = gst_on_fee(charged_fee)
+    settled_on = captured + timedelta(days=2)
+    settlement = Settlement(
+        settlement_id="setl_T1",
+        merchant_id=order.merchant_id,
+        payment_ids=[payment.payment_id],
+        gross_amount=amount,
+        fees=charged_fee,
+        tax=charged_tax,
+        refund_adjustments=0,
+        other_adjustments=0,
+        net_amount=amount - charged_fee - charged_tax - short,
+        settlement_date=settled_on,
+        reference="STL/setl_T1",
+    )
+    bank = BankEntry(
+        bank_entry_id="bnk_T1",
+        amount=settlement.net_amount,
+        credit_or_debit=CreditDebit.CREDIT,
+        date=settled_on,
+        reference=settlement.reference,
+    )
+    refunds = []
+    if orphan_refund_amount:
+        refunds.append(
+            Refund(
+                refund_id="rfnd_T1",
+                payment_id=None,
+                amount=orphan_refund_amount,
+                created_at=T0 + timedelta(days=1),
+                reference=f"ADJ/{customer}/partial",
+            )
+        )
+    return Batch(
+        orders=[order],
+        payments=[payment],
+        refunds=refunds,
+        settlements=[settlement],
+        bank_entries=[bank],
+    )
+
+
+def _case(batch: Batch):
+    cases, _ = reconcile_batch(batch)
+    assert len(cases) == 1
+    return cases[0]
+
+
+def _outcome(batch: Batch, *, use_ai: bool):
+    report = run(batch, use_ai=use_ai, provider=HeuristicProvider() if use_ai else None)
+    assert len(report.outcomes) == 1
+    return report.outcomes[0]
+
+
+# ---- the reported case -------------------------------------------------------
+
+def test_the_reported_shape_is_not_classified_as_a_fee_mismatch():
+    case = _case(_batch(short=SHORT))
+    invariants = {i.name: i for i in case.invariants}
+
+    assert invariants["I2_fees_and_tax_match_schedule"].holds, "fees and tax are on schedule"
+    assert not invariants["I1_settlement_arithmetic_consistent"].holds
+    assert case.exception.exception_type is ExceptionType.UNEXPLAINED_SHORTFALL
+    assert case.exception.evidence["settlement_arithmetic_gap_paise"] == SHORT
+
+
+def test_the_reported_shape_is_never_auto_resolved_rules_only():
+    outcome = _outcome(_batch(short=SHORT), use_ai=False)
+
+    assert outcome.state == HUMAN_REVIEW_REQUIRED
+    assert outcome.state != AUTO_RESOLVED
+    assert "250.00" in outcome.decision.reason
+    assert "own gross, fees, tax, refunds and adjustments" in outcome.decision.reason
+
+
+def test_the_reported_shape_is_never_auto_resolved_with_the_investigator():
+    outcome = _outcome(_batch(short=SHORT), use_ai=True)
+
+    assert outcome.state == HUMAN_REVIEW_REQUIRED
+    assert outcome.decision.state != AUTO_RESOLVED
+
+
+def test_a_matching_orphan_refund_cannot_close_a_settlement_that_does_not_add_up():
+    """The gate would accept this refund. The safety gate must still refuse.
+
+    An unlinked refund of exactly 250.00 for the same customer makes the Shadow
+    Ledger balance, so the Evidence Gate verifies the hypothesis. But the
+    settlement's own components still do not add up to its reported net, so
+    the refund proves nothing about this row.
+    """
+    batch = _batch(short=SHORT, orphan_refund_amount=SHORT)
+    outcome = _outcome(batch, use_ai=True)
+
+    assert outcome.verification is not None
+    assert outcome.verification.verdict == VERIFIED, (
+        "the test is only meaningful if the gate would have accepted the refund"
+    )
+    assert outcome.state == HUMAN_REVIEW_REQUIRED
+    assert outcome.decision.resolution is None
+
+
+def test_components_that_disagree_with_a_correct_net_are_not_closed():
+    """The net is right but a component is not: still an unreconciled row."""
+    batch = _batch()
+    batch.settlements[0].refund_adjustments += SHORT   # net unchanged
+    outcome = _outcome(batch, use_ai=False)
+
+    assert outcome.case.exception.evidence["settlement_arithmetic_gap_paise"] == -SHORT
+    assert outcome.state == HUMAN_REVIEW_REQUIRED
+    assert "above" in outcome.decision.reason
+
+
+# ---- the residual is measured against the reported net -----------------------
+
+def test_residual_is_the_unexplained_money_when_a_fee_overcharge_is_also_present():
+    """A proven fee overcharge plus a short row: the 250.00 must not be netted away."""
+    case = _case(_batch(fee_overcharge=to_paise("40.00"), short=SHORT))
+
+    assert case.exception.exception_type is ExceptionType.FEE_MISMATCH
+    assert case.exception.evidence["residual_paise"] == SHORT
+    outcome = _outcome(_batch(fee_overcharge=to_paise("40.00"), short=SHORT), use_ai=False)
+    assert outcome.state == HUMAN_REVIEW_REQUIRED
+    assert "250.00" in outcome.decision.reason
+
+
+# ---- the rule must not change correct closures -------------------------------
+
+def test_a_clean_fee_overcharge_still_closes_with_zero_residual():
+    batch = _batch(fee_overcharge=to_paise("40.00"))
+    case = _case(batch)
+
+    assert case.exception.exception_type is ExceptionType.FEE_MISMATCH
+    assert case.exception.evidence["residual_paise"] == 0
+    assert case.exception.evidence["settlement_arithmetic_gap_paise"] == 0
+
+    outcome = _outcome(batch, use_ai=False)
+    assert outcome.state == AUTO_RESOLVED
+
+
+def test_a_late_settlement_with_consistent_arithmetic_still_closes():
+    batch = _batch()
+    batch.settlements[0].settlement_date += timedelta(days=5)
+    batch.bank_entries[0].date = batch.settlements[0].settlement_date
+    outcome = _outcome(batch, use_ai=False)
+
+    assert outcome.case.exception.exception_type is ExceptionType.DELAYED_SETTLEMENT
+    assert outcome.state == AUTO_RESOLVED

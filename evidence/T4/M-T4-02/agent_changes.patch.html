+"""A settlement whose reported net does not follow from its own figures.
+
+The rule under test: a proven fee or tax variance may close a case only when
+every rupee the settlement moved is accounted for. A settlement that is short
+of what its own gross, fees, tax and refunds add up to has unexplained money,
+whatever the fee schedule says, so it must escalate rather than auto-close.
+
+The reported case: fees and tax exactly per schedule, reported net INR 250
+short of gross less fees, tax and refunds. The old residual check measured
+against the settlement's components and so reported zero, and the case was
+closed as a fee mismatch.
+"""
+
+from __future__ import annotations
+
+from datetime import datetime, timedelta
+
+from ledgerguard.evidence.safety import AUTO_RESOLVED, HUMAN_REVIEW_REQUIRED
+from ledgerguard.ledger.models import (
+    Batch,
+    BankEntry,
+    CreditDebit,
+    Order,
+    Payment,
+    Settlement,
+)
+from ledgerguard.ledger.money import gst_on_fee, platform_fee, to_paise
+from ledgerguard.pipeline import run
+
+T0 = datetime(2026, 8, 3, 10, 0, 0)
+
+
+def _batch(*, amount: str, settlement_adjust: int = 0, refund_adjustments: int = 0,
+           fee_overcharge: int = 0, net_adjust: int = 0) -> Batch:
+    """One captured payment and its settlement.
+
+    ``fee_overcharge`` moves fee and tax off schedule *consistently* (the net
+    follows), so it is a provable fee mismatch. ``net_adjust`` moves only the
+    reported net, so the settlement contradicts its own components.
+    ``refund_adjustments`` is a deduction on the settlement with no valid refund
+    behind it.
+    """
+    amt = to_paise(amount)
+    fee = platform_fee(amt)
+    tax = gst_on_fee(fee)
+    charged_fee = fee + fee_overcharge
+    charged_tax = gst_on_fee(charged_fee)
+    captured = T0 + timedelta(minutes=5)
+    settled_on = captured + timedelta(days=2)
+    net = amt - charged_fee - charged_tax - refund_adjustments + net_adjust
+    order = Order(
+        order_id="ord_T4", merchant_id="mch_T4", amount=amt,
+        created_at=T0, customer_reference="CUST-T4",
+    )
+    payment = Payment(
+        payment_id="pay_T4", order_id=order.order_id, amount=amt,
+        captured_at=captured, fee=fee, tax=tax, reference="PAY/ord_T4",
+    )
+    settlement = Settlement(
+        settlement_id="setl_T4", merchant_id="mch_T4", payment_ids=[payment.payment_id],
+        gross_amount=amt, fees=charged_fee, tax=charged_tax,
+        refund_adjustments=refund_adjustments, other_adjustments=0,
+        net_amount=net, settlement_date=settled_on, reference="STL/setl_T4",
+    )
+    bank = BankEntry(
+        bank_entry_id="bnk_T4", amount=net, credit_or_debit=CreditDebit.CREDIT,
+        date=settled_on, reference=settlement.reference,
+    )
+    return Batch(orders=[order], payments=[payment], refunds=[],
+                 settlements=[settlement], bank_entries=[bank])
+
+
+def _outcome(batch: Batch):
+    report = run(batch, use_ai=False)
+    assert len(report.outcomes) == 1
+    return report.outcomes[0]
+
+
+def test_short_settlement_with_exact_fees_is_not_auto_closed():
+    """The reported case: fees on schedule, net INR 250 short."""
+    outcome = _outcome(_batch(amount="10000.00", net_adjust=-to_paise("250.00")))
+
+    assert outcome.case.exception.exception_type.value == "EXCEPTION_FEE_MISMATCH"
+    assert outcome.state != AUTO_RESOLVED
+    assert outcome.state == HUMAN_REVIEW_REQUIRED
+
+    evidence = outcome.case.exception.evidence
+    assert evidence["arithmetic_gap_paise"] == to_paise("250.00")
+    assert evidence["residual_paise"] == to_paise("250.00")
+    assert "250.00" in outcome.decision.reason
+    assert "below" in outcome.decision.reason
+    assert "no record" in outcome.decision.reason
+    assert outcome.decision.missing_evidence
+
+
+def test_settlement_over_its_own_components_is_not_auto_closed_either():
+    outcome = _outcome(_batch(amount="10000.00", net_adjust=to_paise("250.00")))
+
+    assert outcome.state == HUMAN_REVIEW_REQUIRED
+    assert outcome.case.exception.evidence["arithmetic_gap_paise"] == -to_paise("250.00")
+    assert "above" in outcome.decision.reason
+

+def test_offsetting_components_do_not_close_even_when_the_net_matches_the_ledger():
+    """The reported net equals the Shadow Ledger's expectation, so a net-only
+    residual would be zero. But the settlement's own deduction is INR 250 more
+    than its net implies. The figures contradict each other, so it escalates.
+    """
+    outcome = _outcome(_batch(amount="10000.00", refund_adjustments=to_paise("250.00"),
+                              net_adjust=to_paise("250.00")))
+
+    evidence = outcome.case.exception.evidence
+    assert evidence["residual_paise"] == 0
+    assert evidence["arithmetic_gap_paise"] == -to_paise("250.00")
+    assert outcome.state == HUMAN_REVIEW_REQUIRED
+

+def test_consistent_fee_overcharge_still_auto_closes_with_no_gap():
+    """Regression: F4 shape. Fee and tax are off schedule and the net follows,
+    so the variance is fully proven and nothing is unexplained."""
+    outcome = _outcome(_batch(amount="10000.00", fee_overcharge=to_paise("40.00")))
+
+    assert outcome.state == AUTO_RESOLVED
+    evidence = outcome.case.exception.evidence
+    assert evidence["residual_paise"] == 0
+    assert evidence["arithmetic_gap_paise"] == 0
+    assert outcome.investigation is None, "a fee mismatch is proven, not investigated"

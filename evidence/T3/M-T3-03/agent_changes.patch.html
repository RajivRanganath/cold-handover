+"""`benchmark --output-dir PATH` writes the same artifacts somewhere else.
+
+Every test redirects OUTPUT_DIR and MANIFEST_DIR into tmp_path, so the real
+evaluation/outputs/ and evaluation/frozen/ are never read-modified by the suite.
+A small seed and count keep each run fast; the stub provider keeps it offline
+and deterministic.
+"""
+
+from __future__ import annotations
+
+import json
+from pathlib import Path
+
+import pytest
+
+from ledgerguard.evaluation import benchmark as bm
+
+SEED = "424242"
+COUNT = "40"
+ARTIFACTS = {"benchmark.md", "benchmark.json", "cases.json", "unresolved_exceptions.csv"}
+TIMING_KEYS = {
+    "wall_seconds",
+    "throughput_per_second",
+    "latency_p50_ms",
+    "latency_p95_ms",
+    "latency_ms",
+}
+TIMING_MD_LABELS = ("Wall time (s)", "Throughput (cases/s)", "Investigation latency")
+
+
+@pytest.fixture
+def sandbox(tmp_path, monkeypatch):
+    """Point the benchmark's default output and manifest locations at tmp_path."""
+    outputs = tmp_path / "sentinel_outputs"  # stands in for evaluation/outputs/
+    frozen = tmp_path / "frozen"
+    frozen.mkdir()
+    monkeypatch.setattr(bm, "OUTPUT_DIR", outputs)
+    monkeypatch.setattr(bm, "MANIFEST_DIR", frozen)
+    return outputs, frozen
+
+
+def _run(*extra: str) -> int:
+    return bm.main(["--seed", SEED, "--count", COUNT, "--provider", "stub", *extra])
+
+
+def _strip_timing(obj):
+    if isinstance(obj, dict):
+        return {k: _strip_timing(v) for k, v in obj.items() if k not in TIMING_KEYS}
+    if isinstance(obj, list):
+        return [_strip_timing(v) for v in obj]
+    return obj
+
+
+def _normalised(path: Path):
+    """Artifact content with only the wall-clock fields removed."""
+    if path.suffix == ".json":
+        return _strip_timing(json.loads(path.read_text()))
+    if path.suffix == ".md":
+        return [
+            line
+            for line in path.read_text().splitlines()
+            if not any(label in line for label in TIMING_MD_LABELS)
+        ]
+    return path.read_bytes()
+
+
+def _written_line(stdout: str) -> str:
+    return next(l for l in stdout.splitlines() if l.startswith("Artifacts written to"))
+
+
+def test_output_dir_writes_the_four_artifacts_and_nothing_else(sandbox, tmp_path, capsys):
+    outputs, frozen = sandbox
+    target = tmp_path / "reports" / "2026" / "run1"  # missing parents on purpose
+
+    assert _run("--output-dir", str(target)) == 0
+
+    assert {p.name for p in target.iterdir()} == ARTIFACTS
+    assert not any(p.name.startswith("holdout_") for p in target.iterdir())
+
+
+def test_output_dir_contents_match_a_default_run(sandbox, tmp_path, capsys):
+    outputs, _frozen = sandbox
+    _run()  # default run, into the sentinel outputs directory
+    capsys.readouterr()
+    target = tmp_path / "custom"
+    _run("--output-dir", str(target))
+    capsys.readouterr()
+
+    assert {p.name for p in outputs.iterdir()} == ARTIFACTS
+    assert {p.name for p in target.iterdir()} == ARTIFACTS
+    for name in sorted(ARTIFACTS):
+        assert _normalised(target / name) == _normalised(outputs / name), name
+
+
+def test_outputs_directory_is_not_created(sandbox, tmp_path):
+    outputs, _frozen = sandbox
+    assert not outputs.exists()
+
+    assert _run("--output-dir", str(tmp_path / "elsewhere")) == 0
+
+    assert not outputs.exists()
+
+
+def test_existing_outputs_files_are_left_byte_identical(sandbox, tmp_path):
+    outputs, _frozen = sandbox
+    outputs.mkdir()
+    sentinel = outputs / "benchmark.md"
+    sentinel.write_text("committed artifact\n")
+    other = outputs / "scale.json"
+    other.write_text("{}\n")
+
+    assert _run("--output-dir", str(tmp_path / "elsewhere")) == 0
+
+    assert sentinel.read_text() == "committed artifact\n"
+    assert other.read_text() == "{}\n"
+    assert sorted(p.name for p in outputs.iterdir()) == ["benchmark.md", "scale.json"]
+
+
+def test_relative_path_resolves_against_current_directory(sandbox, tmp_path, monkeypatch, capsys):
+    work = tmp_path / "work"
+    work.mkdir()
+    monkeypatch.chdir(work)
+
+    assert _run("--output-dir", "rel/out") == 0
+
+    expected = (work / "rel" / "out").resolve()
+    assert {p.name for p in expected.iterdir()} == ARTIFACTS
+    assert f"Artifacts written to {expected}" in capsys.readouterr().out
+
+
+def test_printed_line_names_the_requested_path(sandbox, tmp_path, capsys):
+    target = tmp_path / "named"
+    _run("--output-dir", str(target))
+    line = _written_line(capsys.readouterr().out)
+    assert line == f"Artifacts written to {target.resolve()}"
+
+
+def test_default_run_prints_the_default_outputs_path(sandbox, capsys):
+    outputs, _frozen = sandbox
+    _run()
+    assert _written_line(capsys.readouterr().out) == f"Artifacts written to {outputs}"
+
+
+def test_manifest_is_verified_from_frozen_and_never_written_to_output_dir(sandbox, tmp_path):
+    outputs, frozen = sandbox
+    target = tmp_path / "out"
+
+    assert _run("--output-dir", str(target)) == 0
+    manifests = sorted(frozen.glob("holdout_*.json"))
+    assert len(manifests) == 1  # frozen on first use, as today
+    manifest_bytes = manifests[0].read_bytes()
+
+    # Second run verifies against the frozen manifest and leaves it untouched.
+    assert _run("--output-dir", str(tmp_path / "out2")) == 0
+    assert manifests[0].read_bytes() == manifest_bytes
+    assert sorted(frozen.glob("holdout_*.json")) == manifests
+    assert not list(target.glob("holdout_*")) and not list((tmp_path / "out2").glob("holdout_*"))
+
+
+def test_drifted_frozen_holdout_still_refused_and_writes_nothing(sandbox, tmp_path):
+    _outputs, frozen = sandbox
+    assert _run("--output-dir", str(tmp_path / "first")) == 0
+
+    manifest = next(frozen.glob("holdout_*.json"))
+    data = json.loads(manifest.read_text())
+    data["holdout_batch_sha256"] = "0" * 64
+    manifest.write_text(json.dumps(data, indent=2) + "\n")
+
+    target = tmp_path / "second"
+    with pytest.raises(SystemExit) as excinfo:
+        _run("--output-dir", str(target))
+    assert "FROZEN HOLDOUT MISMATCH" in str(excinfo.value)
+    assert not target.exists() or not any(target.iterdir())
+    assert json.loads(manifest.read_text())["holdout_batch_sha256"] == "0" * 64
+
+
+@pytest.mark.parametrize("subpath", ["", "sub", "sub/deeper"])
+def test_path_inside_outputs_is_refused(sandbox, subpath, capsys):
+    outputs, _frozen = sandbox
+    target = outputs / subpath if subpath else outputs
+
+    with pytest.raises(SystemExit) as excinfo:
+        _run("--output-dir", str(target))
+
+    assert excinfo.value.code == 2
+    assert "reserved for the committed artifacts" in capsys.readouterr().err
+    assert not outputs.exists()

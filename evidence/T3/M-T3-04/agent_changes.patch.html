+"""`benchmark --output-dir PATH`: where artifacts go, and what must not change.
+
+Every test redirects OUTPUT_DIR and MANIFEST_DIR into tmp_path, so the committed
+evaluation/outputs/ and evaluation/frozen/ directories are never written here.
+Runs use the offline stub provider on a small seeded dataset, so they are fast
+and deterministic.
+"""
+
+from __future__ import annotations
+
+import json
+
+import pytest
+
+from ledgerguard.evaluation import benchmark as bm
+
+SEED = "424242"
+COUNT = "40"
+ARTIFACTS = {"benchmark.md", "benchmark.json", "cases.json", "unresolved_exceptions.csv"}
+
+# Wall-clock fields. These are the only values allowed to differ between runs.
+TIMING_KEYS = {"wall_seconds", "throughput_per_second", "latency_p50_ms", "latency_p95_ms", "latency_ms"}
+TIMING_MD_LABELS = ("Wall time", "Throughput", "Investigation latency")
+
+
+@pytest.fixture
+def sandbox(monkeypatch, tmp_path):
+    default_out = tmp_path / "default_outputs"
+    frozen = tmp_path / "frozen"
+    monkeypatch.setattr(bm, "OUTPUT_DIR", default_out)
+    monkeypatch.setattr(bm, "MANIFEST_DIR", frozen)
+    return {"default_out": default_out, "frozen": frozen, "root": tmp_path}
+
+
+def _run(*extra: str) -> int:
+    return bm.main(["--seed", SEED, "--count", COUNT, "--provider", "stub", *extra])
+
+
+def _strip_timing(obj):
+    if isinstance(obj, dict):
+        return {k: _strip_timing(v) for k, v in obj.items() if k not in TIMING_KEYS}
+    if isinstance(obj, list):
+        return [_strip_timing(v) for v in obj]
+    return obj
+
+
+def _normalise_md(text: str) -> str:
+    return "\n".join(
+        line for line in text.splitlines() if not any(lbl in line for lbl in TIMING_MD_LABELS)
+    )
+
+
+def test_output_dir_receives_all_artifacts_and_creates_missing_parents(sandbox, capsys):
+    dest = sandbox["root"] / "a" / "b" / "run"
+    assert _run("--output-dir", str(dest)) == 0
+
+    assert dest.is_dir()
+    assert {p.name for p in dest.iterdir()} == ARTIFACTS
+
+
+def test_output_dir_leaves_default_outputs_untouched(sandbox, capsys):
+    dest = sandbox["root"] / "custom"
+    _run("--output-dir", str(dest))
+
+    assert not sandbox["default_out"].exists()
+
+
+def test_relative_output_dir_resolves_against_cwd(sandbox, monkeypatch, capsys):
+    monkeypatch.chdir(sandbox["root"])
+    assert _run("--output-dir", "rel/out") == 0
+
+    assert {p.name for p in (sandbox["root"] / "rel" / "out").iterdir()} == ARTIFACTS
+    assert not sandbox["default_out"].exists()
+
+
+def test_printed_line_names_the_output_dir(sandbox, capsys):
+    dest = sandbox["root"] / "named"
+    _run("--output-dir", str(dest))
+
+    out = capsys.readouterr().out
+    assert f"Artifacts written to {dest}" in out
+    assert f"Artifacts written to {sandbox['default_out']}" not in out
+
+
+def test_manifest_is_read_from_frozen_dir_and_never_written_to_output_dir(sandbox, capsys):
+    dest = sandbox["root"] / "custom"
+    _run("--output-dir", str(dest))
+
+    manifest = sandbox["frozen"] / f"holdout_{SEED}_{COUNT}.json"
+    assert manifest.exists()
+    assert not any("holdout" in p.name for p in dest.iterdir())
+
+
+def test_frozen_manifest_still_verified_when_output_dir_given(sandbox, capsys):
+    sandbox["frozen"].mkdir(parents=True)
+    bogus = sandbox["frozen"] / f"holdout_{SEED}_{COUNT}.json"
+    bogus.write_text(json.dumps({"holdout_batch_sha256": "0" * 64}))
+    dest = sandbox["root"] / "custom"
+
+    with pytest.raises(SystemExit) as excinfo:
+        _run("--output-dir", str(dest))
+
+    assert "FROZEN HOLDOUT MISMATCH" in str(excinfo.value)
+    assert not dest.exists() or not any(dest.iterdir())
+
+
+def test_refuses_output_dir_inside_committed_outputs_before_any_work(sandbox, monkeypatch, capsys):
+    committed = sandbox["root"] / "committed_outputs"
+    monkeypatch.setattr(bm, "OUTPUT_DIR", committed)
+
+    with pytest.raises(SystemExit):
+        _run("--output-dir", str(committed / "nested"))
+
+    assert not committed.exists()
+    assert not sandbox["frozen"].exists()
+
+
+def test_artifacts_match_default_run_except_wall_clock_fields(sandbox, capsys):
+    custom = sandbox["root"] / "custom"
+    _run()  # default run, into the redirected default directory
+    _run("--output-dir", str(custom))
+
+    default = sandbox["default_out"]
+
+    assert (custom / "unresolved_exceptions.csv").read_bytes() == (
+        default / "unresolved_exceptions.csv"
+    ).read_bytes()
+    assert _strip_timing(json.loads((custom / "benchmark.json").read_text())) == _strip_timing(
+        json.loads((default / "benchmark.json").read_text())
+    )
+    assert _strip_timing(json.loads((custom / "cases.json").read_text())) == _strip_timing(
+        json.loads((default / "cases.json").read_text())
+    )
+    assert _normalise_md((custom / "benchmark.md").read_text()) == _normalise_md(
+        (default / "benchmark.md").read_text()
+    )
+
+
+def test_default_behaviour_unchanged_without_option(sandbox, capsys):
+    assert _run() == 0
+
+    assert {p.name for p in sandbox["default_out"].iterdir()} >= ARTIFACTS
+    assert f"Artifacts written to {sandbox['default_out']}" in capsys.readouterr().out

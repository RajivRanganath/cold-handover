+"""--output-dir for the benchmark.
+
+These run the real benchmark on a small offline dataset (stub provider), with
+OUTPUT_DIR and MANIFEST_DIR redirected into tmp_path, so the committed
+evaluation/outputs/ and evaluation/frozen/ are never read or written.
+"""
+
+from __future__ import annotations
+
+import json
+
+import pytest
+
+from ledgerguard.evaluation import benchmark as bm
+
+SEED = 424242
+COUNT = 40
+ARTIFACTS = ["benchmark.md", "benchmark.json", "cases.json", "unresolved_exceptions.csv"]
+
+# Wall-clock fields. Everything else must match a default run exactly.
+TIMING_KEYS = {"wall_seconds", "throughput_per_second", "latency_p50_ms", "latency_p95_ms", "latency_ms"}
+TIMING_MD_LABELS = ("Wall time (s)", "Throughput (cases/s)", "Investigation latency p")
+
+
+@pytest.fixture
+def sandbox(tmp_path, monkeypatch):
+    default_out = tmp_path / "default_outputs"
+    frozen = tmp_path / "frozen"
+    monkeypatch.setattr(bm, "OUTPUT_DIR", default_out)
+    monkeypatch.setattr(bm, "MANIFEST_DIR", frozen)
+    return {"default_out": default_out, "frozen": frozen, "root": tmp_path}
+
+
+def _run(extra: list[str]) -> int:
+    return bm.main(["--seed", str(SEED), "--count", str(COUNT), "--provider", "stub", *extra])
+
+
+def _strip_timing(obj):
+    if isinstance(obj, dict):
+        return {k: _strip_timing(v) for k, v in obj.items() if k not in TIMING_KEYS}
+    if isinstance(obj, list):
+        return [_strip_timing(v) for v in obj]
+    return obj
+
+
+def test_output_dir_receives_every_artifact(sandbox, capsys):
+    dest = sandbox["root"] / "reports" / "nested"
+    assert _run(["--output-dir", str(dest)]) == 0
+
+    for name in ARTIFACTS:
+        assert (dest / name).is_file(), name
+    assert not sandbox["default_out"].exists()
+
+
+def test_output_dir_relative_path_resolves_against_cwd(sandbox, monkeypatch, capsys):
+    monkeypatch.chdir(sandbox["root"])
+    assert _run(["--output-dir", "rel/out"]) == 0
+
+    dest = (sandbox["root"] / "rel" / "out").resolve()
+    for name in ARTIFACTS:
+        assert (dest / name).is_file(), name
+    assert f"Artifacts written to {dest}" in capsys.readouterr().out
+
+
+def test_printed_line_names_the_given_path(sandbox, capsys):
+    dest = sandbox["root"] / "named"
+    _run(["--output-dir", str(dest)])
+    out = capsys.readouterr().out
+    line = next(l for l in out.splitlines() if l.startswith("Artifacts written to"))
+    assert line == f"Artifacts written to {dest.resolve()}"
+
+
+def test_output_dir_content_matches_default_run_apart_from_timing(sandbox):
+    assert _run([]) == 0
+    assert _run(["--output-dir", str(sandbox["root"] / "alt")]) == 0
+
+    default, alt = sandbox["default_out"], sandbox["root"] / "alt"
+
+    def md_without_timing(path):
+        return [l for l in path.read_text().splitlines() if not any(t in l for t in TIMING_MD_LABELS)]
+
+    assert md_without_timing(default / "benchmark.md") == md_without_timing(alt / "benchmark.md")
+
+    json_default = _strip_timing(json.loads((default / "benchmark.json").read_text()))
+    json_alt = _strip_timing(json.loads((alt / "benchmark.json").read_text()))
+    assert json_default == json_alt
+
+    cases_default = _strip_timing(json.loads((default / "cases.json").read_text()))
+    cases_alt = _strip_timing(json.loads((alt / "cases.json").read_text()))
+    assert cases_default == cases_alt
+
+    assert (default / "unresolved_exceptions.csv").read_text() == (
+        alt / "unresolved_exceptions.csv"
+    ).read_text()
+
+
+def test_output_dir_inside_default_outputs_is_refused(sandbox):
+    sandbox["default_out"].mkdir()
+    for target in (sandbox["default_out"], sandbox["default_out"] / "sub"):
+        with pytest.raises(SystemExit) as excinfo:
+            _run(["--output-dir", str(target)])
+        assert "inside evaluation/outputs/" in str(excinfo.value)
+    assert list(sandbox["default_out"].iterdir()) == []
+    assert not (sandbox["default_out"] / "sub").exists()
+
+
+def test_manifest_is_verified_from_frozen_and_never_written_to_output_dir(sandbox):
+    _run([])  # freezes the manifest for this seed under the sandbox frozen dir
+    manifest = sandbox["frozen"] / f"holdout_{SEED}_{COUNT}.json"
+    before = manifest.read_text()
+
+    dest = sandbox["root"] / "out_with_manifest_check"
+    assert _run(["--output-dir", str(dest)]) == 0
+
+    assert manifest.read_text() == before
+    assert not [p for p in dest.iterdir() if p.name.startswith("holdout")]
+    assert sorted(p.name for p in dest.iterdir()) == sorted(ARTIFACTS)
+
+
+def test_frozen_mismatch_still_aborts_with_output_dir(sandbox):
+    _run([])
+    manifest = sandbox["frozen"] / f"holdout_{SEED}_{COUNT}.json"
+    tampered = json.loads(manifest.read_text())
+    tampered["holdout_batch_sha256"] = "0" * 64
+    manifest.write_text(json.dumps(tampered, indent=2) + "\n")
+
+    dest = sandbox["root"] / "should_stay_empty"
+    with pytest.raises(SystemExit) as excinfo:
+        _run(["--output-dir", str(dest)])
+    assert "FROZEN HOLDOUT MISMATCH" in str(excinfo.value)
+    assert not any(p.exists() for p in (dest / n for n in ARTIFACTS))

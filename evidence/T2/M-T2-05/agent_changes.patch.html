+"""Settlement status: only ``processed`` settlements can settle a payment.
+
+A ``pending`` or ``failed`` settlement has not paid anything out. These tests
+pin that it is invisible to every settlement-matching step (duplicate
+detection, multiple-settlement detection, the invariants, bank matching), that
+a payment left with no processed settlement is reported exactly like a payment
+with no settlement at all, and that the controller never investigates or closes
+such a case.
+
+Nothing here touches the network. The investigator is a spy.
+"""
+
+from __future__ import annotations
+
+from datetime import datetime, timedelta
+
+import pytest
+
+from ledgerguard.ai.provider import HeuristicProvider
+from ledgerguard.evidence.safety import AUTO_RESOLVED, HUMAN_REVIEW_REQUIRED
+from ledgerguard.ledger.models import (
+    BankEntry,
+    Batch,
+    CreditDebit,
+    ExceptionType,
+    Order,
+    Payment,
+    Refund,
+    RefundStatus,
+    Settlement,
+    SettlementStatus,
+)
+from ledgerguard.ledger.money import gst_on_fee, platform_fee
+from ledgerguard.ledger.shadow_ledger import ShadowLedger
+from ledgerguard.pipeline import run
+from ledgerguard.reconciliation.matcher import MATCHED, reconcile_batch
+
+T0 = datetime(2026, 6, 10, 10, 0, 0)
+PROCESSED = SettlementStatus.PROCESSED
+PENDING = SettlementStatus.PENDING
+FAILED = SettlementStatus.FAILED
+AMOUNT = 500000
+
+
+class _SpyProvider(HeuristicProvider):
+    """Counts investigations, delegating to the offline stand-in."""
+
+    def __init__(self) -> None:
+        self.calls = 0
+
+    def investigate(self, context: dict):
+        self.calls += 1
+        return super().investigate(context)
+
+
+def _payment(pid: str = "pay_S", amount: int = AMOUNT, order: bool = True) -> Payment:
+    fee = platform_fee(amount)
+    return Payment(
+        payment_id=pid,
+        order_id=f"ord_{pid}" if order else None,
+        amount=amount,
+        captured_at=T0,
+        fee=fee,
+        tax=gst_on_fee(fee),
+        reference=f"PAY/{pid}",
+    )
+
+
+def _orders(payment: Payment) -> list[Order]:
+    if payment.order_id is None:
+        return []
+    return [
+        Order(
+            order_id=payment.order_id, merchant_id="m", amount=payment.amount,
+            created_at=T0, customer_reference="CUST-S",
+        )
+    ]
+
+
+def _settlement(
+    sid: str,
+    payment: Payment,
+    *,
+    status: SettlementStatus = PROCESSED,
+    net: int | None = None,
+    other_adjustments: int = 0,
+    fees: int | None = None,
+    tax: int | None = None,
+    reference: str | None = None,
+) -> Settlement:
+    fees = payment.fee if fees is None else fees
+    tax = payment.tax if tax is None else tax
+    gross = payment.amount
+    if net is None:
+        net = gross - fees - tax + other_adjustments
+    return Settlement(
+        settlement_id=sid,
+        merchant_id="m",
+        payment_ids=[payment.payment_id],
+        gross_amount=gross,
+        fees=fees,
+        tax=tax,
+        refund_adjustments=0,
+        other_adjustments=other_adjustments,
+        net_amount=net,
+        settlement_date=T0 + timedelta(days=2),
+        status=status,
+        reference=reference or f"STL/{sid}",
+    )
+
+
+def _bank_for(settlement: Settlement) -> BankEntry:
+    return BankEntry(
+        bank_entry_id=f"bnk_{settlement.settlement_id}",
+        amount=settlement.net_amount,
+        credit_or_debit=CreditDebit.CREDIT,
+        date=settlement.settlement_date,
+        reference=settlement.reference,
+    )
+
+
+def _batch(payments, settlements=(), bank=(), refunds=()) -> Batch:
+    orders = [o for p in payments for o in _orders(p)]
+    return Batch(
+        orders=orders,
+        payments=list(payments),
+        settlements=list(settlements),
+        bank_entries=list(bank),
+        refunds=list(refunds),
+    )
+
+
+def _outcome(batch: Batch, pid: str = "pay_S", use_ai: bool = False, provider=None):
+    report = run(batch, use_ai=use_ai, provider=provider)
+    return next(o for o in report.outcomes if o.case.payment_id == pid)
+
+
+def _no_settlement_baseline():
+    payment = _payment()
+    return _outcome(_batch([payment]))

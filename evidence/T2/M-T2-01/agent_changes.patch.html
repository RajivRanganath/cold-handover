+"""Settlement status is respected by every settlement-matching step.
+
+A ``pending`` or ``failed`` settlement has not paid anything out. It must never
+close a payment, never count as a duplicate or conflict, and never be used for
+bank matching or the invariants. A captured payment whose only settlements are
+non-processed is a missing settlement, reported with the non-processed rows as
+evidence so the reviewer can see why.
+
+Nothing here touches the network. The controller test uses a spy provider that
+fails the test if it is ever asked to investigate.
+"""
+
+from __future__ import annotations

+import pytest

+from ledgerguard.ai.schemas import InvestigationResult
+from ledgerguard.evidence.safety import AUTO_RESOLVED, HUMAN_REVIEW_REQUIRED
+from ledgerguard.ledger.models import (
+    Batch,
+    ExceptionType,
+    Settlement,
+    SettlementStatus,
+)
+from ledgerguard.ledger.shadow_ledger import ShadowLedger
+from ledgerguard.pipeline import run
+from ledgerguard.reconciliation.matcher import MATCHED, find_duplicates, reconcile_batch

+@pytest.fixture
+def lc():
+    lifecycles, _ = build_clean_lifecycles(seed=20260905, count=4)
+    return lifecycles[0]

+def _variant(settlement: Settlement, **update) -> Settlement:
+    return settlement.model_copy(update=update, deep=True)
+
+
+def _batch(lc, settlements, bank_entries=None) -> Batch:
+    return Batch(
+        orders=[lc.order],
+        payments=[lc.payment],
+        refunds=list(lc.refunds),
+        settlements=list(settlements),
+        bank_entries=list(lc.bank_entries) if bank_entries is None else list(bank_entries),
+    )
+
+
+def _only_case(batch: Batch):
+    cases, _ = reconcile_batch(batch)
+    assert len(cases) == 1
+    return cases[0]
+
+
+class _SpyProvider:
+    """Records every call. Any call is a failure for the non-AI cases."""
+
+    name = "spy"
+
+    def __init__(self) -> None:
+        self.calls = 0
+
+    def investigate(self, context: dict) -> InvestigationResult:
+        self.calls += 1
+        return InvestigationResult.unavailable("spy: not expected to be asked")

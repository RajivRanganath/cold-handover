+"""Settlement status is respected by reconciliation.
+
+A pending or failed settlement has not paid anything out. It must never be able
+to close a payment: it is ignored by duplicate detection, by the invariants and
+by bank matching, and a payment left with no processed settlement is reported
+as a missing settlement, with the non-processed records attached as evidence.
+
+Nothing here touches the network. The investigator is a spy that records calls.
+"""
+
+from __future__ import annotations
+
+from datetime import datetime, timedelta
+
+from ledgerguard.evidence.safety import AUTO_RESOLVED, HUMAN_REVIEW_REQUIRED
+from ledgerguard.ai.schemas import InvestigationResult
+from ledgerguard.ledger.models import (
+    BankEntry,
+    Batch,
+    CreditDebit,
+    ExceptionType,
+    Order,
+    Payment,
+    Settlement,
+    SettlementStatus,
+)
+from ledgerguard.ledger.money import gst_on_fee, platform_fee
+from ledgerguard.ledger.shadow_ledger import ShadowLedger
+from ledgerguard.pipeline import run
+from ledgerguard.reconciliation.matcher import MATCHED, reconcile_batch
+
+T0 = datetime(2026, 6, 10, 10, 0, 0)
+AMOUNT = 500000
+
+
+def _payment(pid: str = "pay_S1", amount: int = AMOUNT, captured_at=T0) -> Payment:
+    fee = platform_fee(amount)
+    return Payment(
+        payment_id=pid,
+        order_id=f"ord_{pid}",
+        amount=amount,
+        captured_at=captured_at,
+        fee=fee,
+        tax=gst_on_fee(fee),
+        reference=f"PAY/{pid}",
+    )
+
+
+def _order(payment: Payment) -> Order:
+    return Order(
+        order_id=payment.order_id,
+        merchant_id="mch_TEST",
+        amount=payment.amount,
+        created_at=T0,
+        customer_reference="cust_TEST",
+    )
+
+
+def _settlement(
+    sid: str,
+    payment: Payment,
+    status: SettlementStatus = SettlementStatus.PROCESSED,
+    net: int | None = None,
+) -> Settlement:
+    fee = platform_fee(payment.amount)
+    tax = gst_on_fee(fee)
+    consistent_net = payment.amount - fee - tax
+    return Settlement(
+        settlement_id=sid,
+        merchant_id="mch_TEST",
+        payment_ids=[payment.payment_id],
+        gross_amount=payment.amount,
+        fees=fee,
+        tax=tax,
+        refund_adjustments=0,
+        other_adjustments=0,
+        net_amount=consistent_net if net is None else net,
+        settlement_date=T0 + timedelta(days=2),
+        status=status,
+        reference=f"STL/{sid}",
+    )
+
+
+def _bank_for(settlement: Settlement, amount: int | None = None) -> BankEntry:
+    return BankEntry(
+        bank_entry_id=f"bnk_{settlement.settlement_id}",
+        amount=settlement.net_amount if amount is None else amount,
+        credit_or_debit=CreditDebit.CREDIT,
+        date=settlement.settlement_date,
+        reference=settlement.reference,
+    )
+
+
+def _batch(payments, settlements=(), bank=()) -> Batch:
+    orders = [_order(p) for p in payments]
+    return Batch(
+        orders=orders,
+        payments=list(payments),
+        settlements=list(settlements),
+        bank_entries=list(bank),
+    )
+
+
+def _only_case(batch: Batch):
+    cases, _ = reconcile_batch(batch)
+    assert len(cases) == 1
+    return cases[0]
+

+# ---------------------------------- req 7: end-to-end controller never investigates
+class _SpyProvider:
+    name = "spy"
+
+    def __init__(self) -> None:
+        self.calls: list[dict] = []
+
+    def investigate(self, context: dict) -> InvestigationResult:
+        self.calls.append(context)
+        return InvestigationResult.unavailable("spy: should never be called")
+

+def test_end_to_end_never_investigates_or_auto_resolves_non_processed_cases():
+    pending_payment = _payment("pay_PEND")
+    failed_payment = _payment("pay_FAIL")
+    clean_payment = _payment("pay_OK")
+    pending = _settlement("setl_PEND", pending_payment, SettlementStatus.PENDING)
+    failed = _settlement("setl_FAIL", failed_payment, SettlementStatus.FAILED)
+    clean = _settlement("setl_OK", clean_payment)
+    batch = _batch(
+        [pending_payment, failed_payment, clean_payment],
+        [pending, failed, clean],
+        [_bank_for(pending), _bank_for(failed), _bank_for(clean)],
+    )
+

+    report = run(batch, use_ai=True, provider=spy)

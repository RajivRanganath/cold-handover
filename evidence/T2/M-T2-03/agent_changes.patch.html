+"""Settlement status is respected by reconciliation (task T2).
+
+Only a ``processed`` settlement has paid anything out, so only a processed
+settlement may settle a payment. A ``pending`` or ``failed`` settlement is
+ignored by every settlement-matching step: duplicate detection, the invariants,
+bank matching and the multiple-settlement check.
+
+Nothing here touches the network. The investigator is a recording stub.
+"""
+
+from __future__ import annotations
+
+from datetime import datetime, timedelta
+
+import pytest
+
+from ledgerguard.ai.schemas import InvestigationResult
+from ledgerguard.evidence.safety import AUTO_RESOLVED, HUMAN_REVIEW_REQUIRED
+from ledgerguard.ledger.models import (
+    BankEntry,
+    Batch,
+    CreditDebit,
+    ExceptionType,
+    Order,
+    Payment,
+    Settlement,
+    SettlementStatus,
+)
+from ledgerguard.ledger.money import gst_on_fee, platform_fee
+from ledgerguard.ledger.shadow_ledger import ShadowLedger
+from ledgerguard.pipeline import run
+from ledgerguard.reconciliation.matcher import find_duplicates, reconcile_batch
+
+T0 = datetime(2026, 6, 10, 10, 0, 0)
+AMOUNT = 500000
+FEE = platform_fee(AMOUNT)
+TAX = gst_on_fee(FEE)
+NET = AMOUNT - FEE - TAX
+PID = "pay_S01"
+
+
+def _payment() -> Payment:
+    return Payment(
+        payment_id=PID,
+        order_id="ord_S01",
+        amount=AMOUNT,
+        captured_at=T0 + timedelta(minutes=3),
+        fee=FEE,
+        tax=TAX,
+        reference=f"PAY/{PID}",
+    )
+
+
+def _settlement(
+    settlement_id: str,
+    status: SettlementStatus = SettlementStatus.PROCESSED,
+    **overrides,
+) -> Settlement:
+    """A settlement that is internally consistent unless overridden."""
+    fields = dict(
+        settlement_id=settlement_id,
+        merchant_id="mch_S01",
+        payment_ids=[PID],
+        gross_amount=AMOUNT,
+        fees=FEE,
+        tax=TAX,
+        refund_adjustments=0,
+        other_adjustments=0,
+        net_amount=NET,
+        settlement_date=T0 + timedelta(days=2, minutes=3),
+        status=status,
+        reference=f"STL/{settlement_id}",
+    )
+    fields.update(overrides)
+    return Settlement(**fields)
+
+
+def _bank_for(settlement: Settlement, bank_id: str | None = None) -> BankEntry:
+    return BankEntry(
+        bank_entry_id=bank_id or f"bnk_{settlement.settlement_id}",
+        amount=settlement.net_amount,
+        credit_or_debit=CreditDebit.CREDIT,
+        date=settlement.settlement_date,
+        reference=settlement.reference,
+    )
+
+
+def _batch(settlements, bank_entries=None, payments=None) -> Batch:
+    order = Order(
+        order_id="ord_S01",
+        merchant_id="mch_S01",
+        amount=AMOUNT,
+        created_at=T0,
+        customer_reference="CUST-S01",
+    )
+    return Batch(
+        orders=[order],
+        payments=payments if payments is not None else [_payment()],
+        settlements=list(settlements),
+        bank_entries=list(bank_entries or []),
+    )
+
+
+def _outcome_for(batch: Batch, payment_id: str = PID):
+    report = run(batch, use_ai=False)
+    return next(o for o in report.outcomes if o.case.payment_id == payment_id)
+
+
+class RecordingProvider:
+    """An investigator that records every call and would resolve if asked."""
+
+    name = "recording-stub"
+
+    def __init__(self) -> None:
+        self.calls: list[dict] = []
+
+    def investigate(self, context: dict) -> InvestigationResult:
+        self.calls.append(context)
+        return InvestigationResult.unavailable("recording stub: no real investigation")

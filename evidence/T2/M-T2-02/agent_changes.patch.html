+"""Settlement status is respected by reconciliation.
+
+A ``pending`` or ``failed`` settlement has not paid anything out. It must never
+settle a payment, never count as a duplicate or conflict, and never close an
+exception. It may only be cited as evidence when a captured payment has no
+processed settlement at all.
+
+Nothing here touches the network. The investigator is a recording stub.
+"""
+
+from __future__ import annotations

+from ledgerguard.ai.schemas import InvestigationResult
+from ledgerguard.evidence.safety import AUTO_RESOLVED, HUMAN_REVIEW_REQUIRED
+from ledgerguard.ledger.models import (
+    Batch,

+    PaymentStatus,
+    SettlementStatus,

+MISSING = ExceptionType.MISSING_SETTLEMENT.value

+# ----------------------------------------------------------------- helpers
+def _holdout():
+    _dev, holdout = split(build_dataset())
+    return holdout

+def _clean_case(dataset):
+    """A clean payment with exactly one settlement, and that settlement."""
+    fault = dataset.fault_class_by_payment()
+    batch = dataset.batch()
+    for payment in batch.payments:
+        if fault[payment.payment_id] != FaultClass.NONE:
+            continue
+        if payment.status != PaymentStatus.CAPTURED:
+            continue
+        linked = [s for s in batch.settlements if payment.payment_id in s.payment_ids]
+        if len(linked) == 1:
+            return payment, linked[0]
+    raise AssertionError("dataset has no clean single-settlement payment")
+
+
+def _with_settlements(batch: Batch, settlements) -> Batch:
+    return batch.model_copy(update={"settlements": list(settlements)})
+
+
+def _outcome(batch: Batch, payment_id: str, use_ai: bool = False, provider=None):
+    report = run(batch, use_ai=use_ai, provider=provider)
+    return next(o for o in report.outcomes if o.case.payment_id == payment_id)
+
+
+def _scenario(payment, settlements) -> Batch:
+    """A batch holding just one payment and the settlements given for it."""
+    return Batch(payments=[payment], settlements=list(settlements))

+    """Stands in for the investigator and records every call it receives."""

+    name = "recording-stub"

+    def __init__(self) -> None:
+        self.calls: list[dict] = []
+
+    def investigate(self, context: dict) -> InvestigationResult:
+        self.calls.append(context)
+        return InvestigationResult.unavailable("recording stub: no investigation")
